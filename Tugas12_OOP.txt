total_tabungan = 0

print("Masukkan jumlah tabungan Anda setiap bulan selama 12 bulan:")

for bulan in range(1, 13):
  
    tabungan_bulan = float(input(f"Bulan ke-{bulan}: Rp "))
    
    total_tabungan += tabungan_bulan

print("\n" + "="*30)
print(f"Total tabungan Raka selama 1 tahun adalah: Rp {total_tabungan:,.2f}")
print("="*30)
